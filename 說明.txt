using Cathay.Common.Utility;
using Cathay.DataService.DBConn;
using System;
using System.Data;
using System.Data.Common;

/// <summary>
/// PC_Image（線上授信-財力證明）資料存取
/// </summary>
namespace Cathay.Service
{
    public class PcImageDAO
    {
        public PcImageDAO() { }

        /// <summary>
        /// 有則更新、無則新增。
        /// 新增時 IDNO/BranchId/Account/Kind/Type 由 PC_OLAccount 依 SeqNo 帶出。
        /// </summary>
        public int UpsertPC_Image(PcImageVO vo)
        {
            int result;
            try
            {
                using (DbConnection dbConn = DataSetService.GetConnection("connstr") as DbConnection)
                {
                    string sql = @"
IF EXISTS (SELECT 1 FROM PC_Image WHERE SeqNo = @SeqNo)
    UPDATE PC_Image
       SET ImgType = @ImgType, FileId = @FileId, FileExtension = @FileExtension,
           CreateUser = @CreateUser, CreateDate = GETDATE()
     WHERE SeqNo = @SeqNo
ELSE
    INSERT INTO PC_Image (SeqNo, IDNO, BranchId, Account, Kind, Type, ImgType, FileId, FileExtension, CreateUser, CreateDate)
    SELECT SeqNo, IDNO, BranchId, Account, Kind, Type, @ImgType, @FileId, @FileExtension, @CreateUser, GETDATE()
      FROM PC_OLAccount
     WHERE SeqNo = @SeqNo
";
                    System.Collections.ArrayList fNames = new System.Collections.ArrayList();
                    System.Collections.ArrayList fValues = new System.Collections.ArrayList();
                    fNames.Add("SeqNo"); fValues.Add(vo.SeqNo);
                    fNames.Add("ImgType"); fValues.Add(vo.ImgType);
                    fNames.Add("FileId"); fValues.Add(vo.FileId);
                    fNames.Add("FileExtension"); fValues.Add(vo.FileExtension);
                    fNames.Add("CreateUser"); fValues.Add(vo.CreateUser);

                    DbParameter[] dbParams = DataSetService.getTableParams(fNames, fValues, dbConn);
                    result = DBHelper.ExecuteNonQuery(dbConn, CommandType.Text, sql, dbParams);
                }
            }
            catch (Exception ex)
            {
                throw ex;
            }
            return result;
        }
    }
}

用途：依 FileID 向影像系統調閱檔案顯示（只看檔案，不上傳、不異動資料）。
      列表上「信用 / 款項借貸 / 不限 / 借券」檢視連結（ShowPDF）即開啟此頁。

修改：GetImageByFileID 的 switch (Fntype)（:48-82），於 case "C0039" 之後新增：

                case "C0041":
                    FileCategoryName = "財力證明_信用";
                    break;
                case "C0042":
                    FileCategoryName = "財力證明_款項借貸";
                    break;
                case "C0043":
                    FileCategoryName = "財力證明_不限";
                    break;
                case "C0044":
                    FileCategoryName = "財力證明_借券";
                    break;

說明：
  - FileCategoryName 僅用於 PDF 的伺服器暫存檔名與瀏覽器顯示檔名（:84、:135、:141），
    不影響調閱哪一個影像（調閱只依 FileID）。
  - 未補之前，C0041～C0044 的檔名只剩「分公司代號＋使用者代號」，仍可正常檢視。
  - 只新增 case，其他呼叫端（IASMAT0061、IASMAT0121、IASMAT0251、IASQRY0161、QueryCreditActsCredentials、
    ModifyCreditActStatus 票查）使用的類別不受影響。


==========================================================================================================
待確認 / 測試注意事項
==========================================================================================================
1. 影像系統是否接受「JobItemID=1 + DetailNo=C0041～C0044」：這四個代碼在程式中僅用於列表檢視，從未實際上傳過，
   程式碼無法確認。可查 CGconnstr 連線資料庫的 LapTable（開戶類別 1 下是否有這四個文件類別），或直接於測試環境上傳一次確認。
2. IM003 只處理 .pdf / .jpg / .gif / .png（:126-150）；上傳頁允許的 .bmp / .tif / .txt 上傳後點檢視會是空白頁。
   是否將上傳頁限縮為 pdf / jpg / gif，另行決定。
3. IM003 的 PDF 暫存檔名不含 FileID，同一使用者、同分公司、同類別會覆寫同一暫存檔，同時檢視時有機會看到別份檔案
   （既有設計，本次不處理）。
4. IM001Handle.ashx 為共用 Handler，請回歸測試既有路徑：
   - 線上授信審件 上傳票查（C0040 + Type=C → PC_RPA）
   - 證券-線上額度調整審件 上傳票查（C0040 → OnlineLimitUpload）
   - 其他使用 IM001.aspx 的上傳功能
5. 同時上傳（併發）不處理；同一 SeqNo 重新上傳會覆蓋 PC_Image 該筆並更新 CreateDate（財力證明日期）。
